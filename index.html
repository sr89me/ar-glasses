function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("Đơn Hàng") || ss.insertSheet("Đơn Hàng");
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["Mã Đơn", "Thời Gian", "Tên Khách", "SĐT", "Địa Chỉ", "Sản Phẩm", "Giá Bán", "Thanh Toán", "Ghi Chú"]);
  }
  var data = JSON.parse(e.postData.contents);
  sheet.appendRow([data.orderCode, data.timestamp, data.name, data.phone, data.address, data.glassName, data.glassPrice, data.payment, data.note]);
  return ContentService.createTextOutput(JSON.stringify({status: "ok"})).setMimeType(ContentService.MimeType.JSON);
}
